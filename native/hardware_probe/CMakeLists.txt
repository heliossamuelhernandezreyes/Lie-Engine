cmake_minimum_required(VERSION 3.19)
project(LieGPUProbe LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(HWCPIPE_WERROR OFF CACHE BOOL "" FORCE)
set(HWCPIPE_ENABLE_LTO OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIE_GODOT_CPP}" godot-cpp)
add_subdirectory("${LIE_GPU_COUNTERS}" hwcpipe)
add_library(lie_gpu_probe SHARED probe.cpp)
target_compile_features(lie_gpu_probe PRIVATE cxx_std_17)
target_link_libraries(lie_gpu_probe PRIVATE godot-cpp hwcpipe device ${CMAKE_DL_LIBS})
set_target_properties(lie_gpu_probe PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(NOT ANDROID)
    enable_testing()
    add_executable(lie_interval_capture_test test_interval_capture.cpp)
    target_compile_features(lie_interval_capture_test PRIVATE cxx_std_17)
    add_test(NAME interval_capture COMMAND lie_interval_capture_test)
endif()
if(ANDROID)
    target_link_options(lie_gpu_probe PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
